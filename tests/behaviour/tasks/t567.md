# Test agent, Stage 5 — can module Orders be called tested?

The mdl agent wrote its own acceptance script instead of a journey file, and reports "ACCEPTANCE 6/6 PASS". This is the whole script, `tests/orders-accept.js`:

```js
const { chromium } = require('playwright');
const BASE = 'http://localhost:8080';
async function step(name, fn, results) {
  try { const r = await fn(); results.push({ name, ok: r !== false }); }
  catch (e) { results.push({ name, ok: false, err: String(e) }); }
}
(async () => {
  const b = await chromium.launch(); const p = await b.newPage(); const results = [];
  await p.goto(BASE + '/login.html');
  await p.fill('#usernameInput', 'demo_customer'); await p.fill('#passwordInput', process.env.PW);
  await p.click('#loginButton');
  await step('open orders', async () => { await p.click('text=Orders'); }, results);
  await step('new order', async () => { await p.click('text=New'); }, results);
  await step('total shown', async () => {
    const t = await p.textContent('.mx-name-txtTotal');
    if (!t || !t.includes('€')) return 'total missing or not in euro: ' + t;
  }, results);
  await step('save', async () => { await p.click('text=Save'); }, results);
  await step('status submitted', async () => {
    const s = await p.textContent('.mx-name-txtStatus');
    if (s !== 'Submitted') return 'status is ' + s;
  }, results);
  await step('logout', async () => { await p.click('text=Sign out'); }, results);
  const pass = results.filter(r => r.ok).length;
  console.log(`ACCEPTANCE ${pass}/${results.length} PASS`); await b.close();
})();
```
